\begin{frame}{Review: SI Characterization Theorem}

      \fig{.80}{figs/si-sigmod1995.png}
    \end{column}
    \begin{column}{.57\textwidth}\centering

        An item-operation history satisfies SI iff its dependency graph is

      \end{theorem}
    \end{column}
  \end{columns}\vfill
