\begin{frame}{Review: SI Characterization Theorem}

    \end{center}

    \column{0.57\textwidth}
    \begin{center}
      \begin{theorem}
        An item-operation history satisfies SI iff its dependency graph is
        internally consistent and every cycle has two adjacent $\RW$ edges.
      \end{theorem}
    \end{center}
